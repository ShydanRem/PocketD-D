set(QJS "${CMAKE_CURRENT_LIST_DIR}/../../.deps/quickjs")
if(NOT EXISTS "${QJS}/quickjs.c")
    message(FATAL_ERROR "Pinned QuickJS missing: run npm run firmware:deps")
endif()
include(${CMAKE_CURRENT_LIST_DIR}/prepare.cmake)
pocket_prepare_quickjs("${QJS}" "${CMAKE_CURRENT_BINARY_DIR}/quickjs-pocket.c")
find_program(POCKET_NODE node REQUIRED)
execute_process(COMMAND "${POCKET_NODE}" "${CMAKE_CURRENT_LIST_DIR}/../../../scripts/build-beyond-bundle.js" --check RESULT_VARIABLE STALE)
if(STALE)
    message(FATAL_ERROR "Beyond bundle stale: run node scripts/build-beyond-bundle.js")
endif()
idf_component_register(SRCS "${CMAKE_CURRENT_BINARY_DIR}/quickjs-pocket.c" "${QJS}/dtoa.c" "${QJS}/libregexp.c" "${QJS}/libunicode.c" "${QJS}/cutils.c" INCLUDE_DIRS "${QJS}")
target_compile_definitions(${COMPONENT_LIB} PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04-pocket")
target_compile_options(${COMPONENT_LIB} PRIVATE -fwrapv -Wno-error=format -Wno-array-bounds -Wno-unused-parameter)
